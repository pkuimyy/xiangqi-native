cmake_minimum_required(VERSION 3.23)
project(XiangqiNative VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/MsvcIncludePrefix.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(external/quickjs EXCLUDE_FROM_ALL)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(external/oxqf EXCLUDE_FROM_ALL)
add_library(xiangqi_core STATIC src/core/kernel.cpp src/core/document.cpp src/core/files.cpp src/core/uci.cpp)
set_source_files_properties(src/core/kernel.cpp PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/kernel/generated/kernel.inc")
target_include_directories(xiangqi_core PUBLIC include external/json PRIVATE kernel/generated)
target_link_libraries(xiangqi_core PUBLIC OXQF::core OXQF::convert PRIVATE qjs)
if(WIN32)
 target_link_libraries(xiangqi_core PRIVATE bcrypt)
endif()
add_executable(xiangqi_cli src/cli/main.cpp)
target_link_libraries(xiangqi_cli PRIVATE xiangqi_core)
enable_testing()
add_executable(xiangqi_core_tests tests/core_tests.cpp)
target_link_libraries(xiangqi_core_tests PRIVATE xiangqi_core)
add_test(NAME native.core COMMAND xiangqi_core_tests)
add_executable(fake_uci tests/fake_uci.cpp)
add_executable(xiangqi_engine_tests tests/engine_tests.cpp)
target_link_libraries(xiangqi_engine_tests PRIVATE xiangqi_core)
add_test(NAME native.engine COMMAND xiangqi_engine_tests $<TARGET_FILE:fake_uci>)
